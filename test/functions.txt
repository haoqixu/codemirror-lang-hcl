# Function call

result = upper("hello")

==>

ConfigFile(Body(
  Attribute(
    Identifier,
    AssignOp,
    Expression(
      FunctionCall(
        FunctionName(Identifier),
        FunctionArguments(
          Expression(
            LiteralValue(
              StringLit(
                QuotedTemplateStart,
                TemplateLiteral,
                QuotedTemplateEnd)))))))))

# Namespaced function call

result = provider::function("arg")

==>

ConfigFile(Body(
  Attribute(
    Identifier,
    AssignOp,
    Expression(
      FunctionCall(
        FunctionNamespace(
          Identifier,
          NamespaceSeparator),
        FunctionName(Identifier),
        FunctionArguments(
          Expression(
            LiteralValue(
              StringLit(
                QuotedTemplateStart,
                TemplateLiteral,
                QuotedTemplateEnd)))))))))

# Namespaced function call with whitespace

result = provider :: function()

==>

ConfigFile(Body(
  Attribute(
    Identifier,
    AssignOp,
    Expression(
      FunctionCall(
        FunctionNamespace(
          Identifier,
          NamespaceSeparator),
        FunctionName(Identifier))))))

# Namespaced function call in template interpolation

result = "${provider::function(var.input)}"

==>

ConfigFile(Body(
  Attribute(
    Identifier,
    AssignOp,
    Expression(
      TemplateExpr(
        QuotedTemplate(
          QuotedTemplateStart,
          TemplateInterpolation(
            TemplateInterpolationStart,
            Expression(
              FunctionCall(
                FunctionNamespace(
                  Identifier,
                  NamespaceSeparator),
                FunctionName(Identifier),
                FunctionArguments(
                  Expression(
                    VariableExpr(Identifier),
                    GetAttr(DerefOp,Identifier))))),
            TemplateInterpolationEnd),
          QuotedTemplateEnd))))))
