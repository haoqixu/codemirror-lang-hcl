# Object attributes

value = {
  name = "example"
  enabled: true,
}

==>

ConfigFile(
  Body(
    Attribute(
      Identifier,
      AssignOp,
      Expression(
        CollectionValue(
          Object(
            ObjectElem(
              ObjectKey(Expression(VariableExpr(Identifier))),
              AssignOp,
              Expression(LiteralValue(StringLit(QuotedTemplateStart,TemplateLiteral,QuotedTemplateEnd)))
            ),
            ObjectElem(ObjectKey(Expression(VariableExpr(Identifier))),Colon,Expression(LiteralValue(BoolLit(true)))),
            Comma
          )
        )
      )
    )
  )
)

# Dynamic object key

value = {
  (var.key) = var.value
}

==>

ConfigFile(
  Body(
    Attribute(
      Identifier,
      AssignOp,
      Expression(
        CollectionValue(
          Object(
            ObjectElem(
              ObjectKey(Expression(Expression(VariableExpr(Identifier),GetAttr(DerefOp,Identifier)))),
              AssignOp,
              Expression(VariableExpr(Identifier),GetAttr(DerefOp,Identifier))
            )
          )
        )
      )
    )
  )
)
